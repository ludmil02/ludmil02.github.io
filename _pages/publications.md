(function () {
  var table = document.getElementById('pubs');
  if (!table) return;
  var tbody = table.tBodies[0];
  var heads = table.tHead.rows[0].cells;
  var rows  = Array.prototype.slice.call(tbody.rows);

  function val(tr, k, t) {
    var d = tr.cells[k].getAttribute('data-v');
    return t === 'n' ? parseFloat(d) : String(d);
  }

  function sortBy(k, dir) {
    var t = heads[k].getAttribute('data-t');
    var sorted = rows.slice().sort(function (a, b) {
      var x = val(a, k, t), y = val(b, k, t), c;
      if (t === 'n') c = (x === y) ? 0 : (x < y ? -1 : 1);
      else c = x.localeCompare(y);
      if (c === 0) c = val(a, 0, 'n') - val(b, 0, 'n');   // stable tiebreak on #
      return dir === 'ascending' ? c : -c;
    });
    var frag = document.createDocumentFragment();
    sorted.forEach(function (r) { frag.appendChild(r); });
    tbody.appendChild(frag);
    for (var i = 0; i < heads.length; i++) heads[i].removeAttribute('aria-sort');
    heads[k].setAttribute('aria-sort', dir);
  }

  for (var i = 0; i < heads.length; i++) {
    (function (k) {
      var th = heads[k];
      th.setAttribute('tabindex', '0');
      th.setAttribute('role', 'columnheader');
      function go() {
        var cur = th.getAttribute('aria-sort');
        // numeric columns open descending (most cites / newest first)
        var dflt = th.getAttribute('data-t') === 'n' && k !== 0 ? 'descending' : 'ascending';
        var dir = cur === 'ascending' ? 'descending' : (cur === 'descending' ? 'ascending' : dflt);
        sortBy(k, dir);
      }
      th.addEventListener('click', go);
      th.addEventListener('keydown', function (e) {
        if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); go(); }
      });
    })(i);
  }

  var q = document.getElementById('q'), count = document.getElementById('count');
  function filter() {
    var s = q.value.trim().toLowerCase(), shown = 0;
    rows.forEach(function (r) {
      var hit = !s || r.textContent.toLowerCase().indexOf(s) !== -1;
      r.style.display = hit ? '' : 'none';
      if (hit) shown++;
    });
    count.textContent = s ? shown + ' of ' + rows.length + ' shown' : '';
  }
  q.addEventListener('input', filter);
})();
